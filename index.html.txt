<script>
let noCount = 0;
let noScale = 1;
let timerStarted = false;
let timeLeft = 5;

function moveNo() {
  const no = document.getElementById("no");
  no.style.left = Math.random() * (window.innerWidth - 120) + "px";
  no.style.top = Math.random() * (window.innerHeight - 120) + "px";
}

function evilNo() {
  noCount++;
  document.body.classList.add("shake");
  setTimeout(() => document.body.classList.remove("shake"), 400);

  noScale -= 0.2;
  document.getElementById("no").style.transform = `scale(${noScale})`;

  if (noCount === 1) {
    showAlert("alert1");
    document.getElementById("message").innerText = "😐 You already hovered…";
  } 
  else if (noCount === 2) {
    showAlert("alert2");
    document.getElementById("message").innerText = "🤨 Screenshot attempt noted.";
  } 
  else if (noCount === 3) {
    showAlert("alert3");
    document.getElementById("message").innerText = "😈 You agreed yesterday actually.";
    startCountdown();
  } 
  else {
    yesClicked();
  }
}

function showAlert(id) {
  document.getElementById(id).style.display = "block";
}

function closeAlert(id) {
  document.getElementById(id).style.display = "none";
}

function startCountdown() {
  if (timerStarted) return;
  timerStarted = true;
  const cd = document.getElementById("countdown");

  const interval = setInterval(() => {
    cd.innerText = `Auto-confirming YES in ${timeLeft}… 💖`;
    timeLeft--;
    if (timeLeft < 0) {
      clearInterval(interval);
      yesClicked();
    }
  }, 1000);
}

function yesClicked() {
  document.body.innerHTML = `
    <h1>💖 VALENTINE CONFIRMED 💖</h1>
    <div id="receipt">
      <h3>💌 Valentine Receipt</h3>
      <p>Name: Chi</p>
      <p>Status: ACCEPTED 😌</p>
      <p>Compatibility: 99.9%</p>
      <p>Cancellation: Not Available</p>
      <p style="margin-top:10px;">🎀 Thank you for choosing love 🎀</p>
    </div>
  `;
  document.getElementById("receipt").style.display = "block";
}
